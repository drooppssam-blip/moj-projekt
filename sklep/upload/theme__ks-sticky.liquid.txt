<script>
  (function () {
    var bar = document.getElementById('ks-{{ section.id }}');
    if (!bar || bar.dataset.ksInit) return;
    bar.dataset.ksInit = '1';
    function setVisible(show) {
      bar.classList.toggle('is-visible', show);
      bar.setAttribute('aria-hidden', show ? 'false' : 'true');
      var b = bar.querySelector('button');
      if (b) b.tabIndex = show ? 0 : -1;
    }
    function watch() {
      var target = document.querySelector('[data-ks-buy]');
      if (!target || !('IntersectionObserver' in window)) { setVisible(true); return; }
      new IntersectionObserver(function (entries) {
        var e = entries[0];
        setVisible(!e.isIntersecting && e.boundingClientRect.top < 0);
      }).observe(target);
    }
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', watch); else watch();
  })();
</script>
